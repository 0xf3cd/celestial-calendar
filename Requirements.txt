# `requests` reaches CI through `automation/github.py`, which only `toolbox/*_downloader.py`
# import -- that is `release.yml` collecting the artifacts, not the build and test legs. A break
# there does not turn a leg red, it fails a release; pinned because that is the worse day to
# discover a dependency moved, not because it decides a verdict.
requests==2.34.2
# The AI-workflow gate reads `.github/workflows/*.yml` for real: a per-job permission and a
# pinned SHA cannot be told apart from a workflow's own prose by matching lines (#144).
# That makes this parser part of a gate, not a convenience -- a silent major would change what
# the gate accepts on a day nobody touched the repository.
pyyaml==6.0.3
# `automation/tests/test_markdown.py` reads links, anchors, code blocks and sections, and the closeout
# gate's license checks (the package READMEs and the CHANGELOG entry) read license sections and
# links, all from one CommonMark parse
# (`automation/markdown_source.py`): a link in a code span or a comment, or a fence that opens a list
# item, cannot be told from what a reader sees by matching lines (#328, #350). Pinned like pyyaml.
markdown-it-py==4.2.0
# The automation layer's own unit tests (#87): pytest runs `automation/tests/` via
# `checks.py --pytest`. It is a gate like any other here -- a silent major would change
# what the gate accepts on a day nobody touched the repository.
pytest==9.1.1
